% Computational graph: deterministic core + generative residual flow.
% Self-contained tikzpicture (styles inline); requires the kugreen* colors and
% \usetikzlibrary{arrows.meta, positioning, calc}. Shared by presentation.tex
% (Model frame) and standalone.tex (PNG export).
\begin{tikzpicture}[
  font=\small,
  state/.style={draw=kugreen, fill=kugreenlyslyslys, rounded corners=2pt,
                minimum height=0.72cm, inner sep=4pt},
  model/.style={draw=kugreen, fill=kugreen, text=white, rounded corners=2pt,
                minimum height=0.9cm, inner sep=5pt},
  emb/.style={draw=kugreenlys, fill=kugreenlyslyslys!40, rounded corners=2pt,
              inner sep=3pt, font=\scriptsize},
  op/.style={draw=kugreen, circle, fill=white, inner sep=1pt},
  flow/.style={-{Latex[length=2.2mm]}, kugreen!80!black, thick},
  loop/.style={-{Latex[length=2.2mm]}, kugreen!80!black, thick, dashed},
  lbl/.style={font=\scriptsize, text=kugreen!60!black},
]

% ===== lane 1: deterministic pipeline (inputs on the same row) =====
\node[state] (xprev) at (-0.3,3.0) {$x_{n-1}$};
\node[state] (xcur)  at (0.8,3.0) {$x_{n}$};
\node[op]    (catdet) at (1.75,3.0) {$\|$};
\node[model] (det)   at (3.9,3.0) {deterministic model};
\node[state] (xdet)  at (6.6,3.0) {$\hat{x}^{\mathrm{det}}_{n+1}$};

\draw[flow] (xprev.north) to[out=40, in=95, looseness=0.9] (catdet.north);
\draw[flow] (xcur) -- (catdet);
\draw[flow] (catdet) -- (det);
\draw[flow] (det) -- (xdet);

% ===== time embeddings (month/hour shared by both models; emb_t only for gen) =====
% \mkern separates the subscript from the "b"; \mathrm subscripts (not beamer's
% bold-ish sans \text) keep the weight uniform
\node[emb] (temb) at (3.9,1.7)
  {$\mathrm{emb}_{\mkern3mu\mathrm{month}}+\mathrm{emb}_{\mkern3mu\mathrm{hour}}$};
\node[emb] (et)   at (4.6,0.42) {$\mathrm{emb}_{\mkern3mu t}$};
\draw[flow] (temb) -- (det);

% ===== lane 2: residual flow (T Euler steps) =====
\node[state] (z0) at (-4.7,-0.8) {$z^{0}\!\sim\!\mathcal{N}(0,\mathbf{I})$};
\node[state] (zi) at (-2.3,-0.8) {$z^{t}$};
\node[state] (catgen) at (0.6,-0.8)
  {$\bigl[\, z^{t} \,\|\, x_{n-1} \,\|\, x_{n} \,\|\, \hat{x}^{\mathrm{det}}_{n+1} \,\bigr]$};
\node[model] (gen) at (3.9,-0.8) {flow model};
\node[state] (ri) at (6.4,-0.8) {$\hat{r}_t$};
\node[state] (ui) at (6.4,-2.5) {$u_t=\dfrac{\hat{r}_t-z^{t}}{s_t}$};
\node[state] (euler) at (2.6,-2.5) {$z^{t+1}=z^{t}+h_t\,u_t$};

\draw[flow] (temb) -- (gen);
\draw[flow] (et.south) -- ($(gen.north)+(0.7,0)$);
\draw[flow] (z0) -- node[lbl, above] {$t=0$} (zi);
\draw[flow] (zi) -- (catgen);
\draw[flow] (catgen) -- (gen);
\draw[flow] (gen) -- (ri);
\draw[flow] (ri) -- (ui);
\draw[flow] (ui) -- (euler);
\draw[loop] (euler.west) -| node[lbl, pos=0.25, below] {repeat $t=0,\dots,T\!-\!1$} (zi.south);

% conditioning wires into the concatenation (the det output is re-used!)
\draw[flow] (xprev.south) -- (catgen.north -| xprev.south);
\draw[flow] (xcur.south)  -- (catgen.north -| xcur.south);
\draw[flow, rounded corners=4pt] (xdet.south) |- (1.8,0.85)
  -- ($(catgen.north)+(1.2,0)$);

% ===== composition =====
\node[op]    (plus)   at (10.9,3.0) {$\oplus$};
\node[state] (xfinal) at (13.3,3.0) {$\hat{x}_{n+1}=\hat{x}^{\mathrm{det}}_{n+1}+\hat{r}_{n+1}$};
\node[state] (zT)     at (10.9,-3.7) {$z^{T}=:\hat{r}_{n+1}$};

\draw[flow] (euler.south) |- node[lbl, pos=0.75, below] {after $T$ steps} (zT.west);
\draw[flow] (zT.north) -- (plus.south);
\draw[flow] (xdet.east) -- (plus.west);
\draw[flow] (plus) -- (xfinal);

% per-step loss estimates (orange), enabled by defining \ShowLossEstimates
% before \input: at EVERY flow step t, jumping straight out of the Euler
% update gives r_t (the estimate of the final residual) and with it the
% clean-state estimate x_t that the guidance loss scores.
\ifdefined\ShowLossEstimates
  \node[state, draw=orange!80!black, fill=orange!15] (rto) at (10.9,-4.5)
    {$\hat{r}_t$};
  \node[state, draw=orange!80!black, fill=orange!15] (xto) at (13.3,1.85)
    {$\hat{x}_{t}=\hat{x}^{\mathrm{det}}_{n+1}+\hat{r}_t$};
  \draw[flow, orange!80!black] ($(euler.south)+(0.6,0)$) |-
    node[pos=0.75, below, font=\scriptsize, text=orange!80!black]
    {at every step $t$} (rto.west);
  \draw[flow, orange!80!black] (rto.east) -| (xto.south);
\fi

% caption line (drops lower when the orange overlay needs the space)
\ifdefined\ShowLossEstimates
  \node[lbl, anchor=north] at (3.7,-5.6)
    {$s_t$: noise level \quad $h_t$: flow step size};
\else
  \node[lbl, anchor=north] at (3.7,-4.5)
    {$s_t$: noise level \quad $h_t$: flow step size};
\fi

\end{tikzpicture}%
