\section{Introduction}

\frame{
\frametitle{Critical systems and extreme weather}
\begin{itemize}
  \item Energy, transport, agriculture: weather dependent
  \begin{itemize}
    \item Short-to-medium-term operation $\leftarrow$ forecasting
    \item Long-term planning $\leftarrow$ scenario analysis (including extremes)
  \end{itemize}
  \item Extremes: unlikely but highly relevant for decision-making
  \item Analyze consequences $\rightarrow$ map bottlenecks \& points of failure
\end{itemize}
\vspace{1em}
$\rightarrow$ \textbf{How can these scenarios be generated?}
}

\frame{
\frametitle{High-level objective}
\textbf{Objective:} simulate \emph{extreme}, yet \emph{plausible} weather scenarios
\begin{itemize}
  \item Extreme + plausible == usability requirements
  \item Scenarios: simulate at different intensity levels
  \item Example: 
  \begin{itemize}
    \item if the average temperature 
    \item over the next $N$ days 
    \item over this region
    \item goes up by $2^{\circ}$, $3^{\circ}$, $4^{\circ}$
    \item $\rightarrow$ then \dots
  \end{itemize}
\end{itemize}
\vspace{1em}
\textbf{Definitions:}
\begin{itemize}
  \item \emph{Extreme}
  \begin{itemize}
    \item unlikely ($\rightarrow$ difficult to predict) and disruptive
    \item tails of the climatological (and forecasting) distribution
  \end{itemize}
  \item \emph{Plausible}
  \begin{itemize}
    \item spatial- and pressure-level change
    \item variables co-variation
  \end{itemize}
\end{itemize}
}

\frame{
\frametitle{Sub-objectives and building blocks}
\textbf{We want:}
\begin{itemize}
  \item a \textbf{practical} algorithm for generating extreme scenarios
  \item scenarios can be \textbf{explicitly specified}
  \item \textbf{low} compute and memory cost
\end{itemize}
\vfill
\textbf{We combine two ingredients:}
\vspace{0.4em}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{1.\ Base model}
\begin{itemize}
  \item \textit{ArchesWeather}
  \item medium-range generative\\probabilistic forecasts
  \item flow matching
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{2.\ Guidance methods}
\begin{itemize}
  \item \textit{classifier-free guidance}
  \item borrowed from image generation domain
  \item \textbf{adapted to} weather domain
\end{itemize}
\end{column}
\end{columns}
\vfill
}

\frame{
\frametitle{Images vs.\ extreme weather}
What and why \textbf{adapt}?

\vspace{1em}
$\rightarrow$ 3 key differences:
\begin{itemize}
  \item \textbf{Steps:}
  \begin{itemize}
    \item images $\rightarrow$ final image obtained one-shot
    \item weather $\rightarrow$ \textbf{multi-step trajectory} (video-alike, but not really \dots)
  \end{itemize}
  \item \textbf{Target:}
  \begin{itemize}
    \item image $\rightarrow$ inherent to a specific task (ground-truth available)
    \item weather $\rightarrow$ unclear a priori (\textbf{designed})
  \end{itemize}
  \item \textbf{Evaluation:}
  \begin{itemize}
    \item image $\rightarrow$ visual appeal
    \item weather $\rightarrow$ qualitative (visual) $+$ quantitative \textbf{plausibility} tests
  \end{itemize}
\end{itemize}

\frame{
\frametitle{Classifier-free guidance}

\textbf{Steering with a loss gradient $=$ sampling from a posterior}
\vspace{-0.4em}
\begin{gather*}
p(z^t|y) \propto p(y|z^t)\,p(z^t)\\
\nabla_{z^t}\log p(z^t|y)
=
\nabla_{z^t}\log p(z^t)
+
\nabla_{z^t}\log p(y|z^t)
\end{gather*}
\vspace{-1.0em}
\begin{itemize}
    \item define likelihood: $p(y|x) \propto e^{-\mathcal{L}(x)}$ where $x:=\hat{x}_T$
\end{itemize}
\vspace{-0.8em}
\begin{gather*}
p(y|z^t) = \int p(y|x)\, p(x|z^t)\, dx
         = \mathbb{E}_{x|z^t}\!\left[\, e^{-\mathcal{L}(x)} \right]
\approx e^{-\mathcal{L}(\hat{x}_t)}
\end{gather*}
\vspace{-1.0em}
\begin{block}{Guided vector field}
\vspace{-0.6em}
\begin{gather*}
\tilde{u}_t = u_t - \lambda_t\, g_t, \quad \text{with } \lambda_t := w \cdot a_t \cdot c_t
\\
g_t = \nabla_{z^{t}} \mathcal{L}(\hat{x}_t) = -\nabla_{z^{t}}\log p(y|z^t)
\end{gather*}
\end{block}
\vspace{-0.4em}
\begin{itemize}
    \item exact posterior needs $w=1$ and $c_t=1$ with path-specific $a_t$
          \\(e.g.\ $\tfrac{s_t}{1-s_t}$ for linear gaussian probability paths)
    \item we trade exactness for \emph{control} by nudging $\lambda_t = w \cdot a_t \cdot c_t$
\end{itemize}
}

\frame{
\frametitle{Adapted classifier-free guidance}
At every flow step $t$:
\begin{enumerate}
  \item Estimate the clean state from the current noisy residual:
        $\hat{x}_t = \hat{x}^{\text{det}} + \hat{r}_t$
  \item Score it with the masked loss: $\mathcal{L}(\hat{x}_t)$
  \item Take the gradient w.r.t.\ the noisy state:
        $g_t = \nabla_{z^{t}} \mathcal{L}(\hat{x}_t)$
  \item Modify the velocity: $\tilde{u}_t = u_t - \lambda_t\, g_t$ with $\lambda_t = w \cdot a_t \cdot c_t$
  \item Take Euler step as usual: $z^{t+1} = z^{t} + h_t\, \tilde{u}_t$
\end{enumerate}

\frame{
\frametitle{Loss}
\vfill
\begin{align*}
  M(x) &\;=\; \sum m \odot x
        \qquad\text{\footnotesize masked average}\\[6pt]
  \mathcal{L}    &\;=\; \Big(\, \underbrace{M(\hat{x}_t)}_{\text{current}}
        \;-\; \underbrace{(1+\phi_n)\,M(x^{\text{ref}})}_{\text{target}} \,\Big)^{2}
        \;=\; e_t^{\,2}
\end{align*}
\vfill
\begin{center}
\begin{tabular}{@{}r@{\quad}l@{}}
  $\hat{x}_t$      & clean-state estimate at flow step $t$ (where we are)\\[3pt]
  $x^{\text{ref}}$ & unguided forecast (or ground truth)\\[3pt]
  $\phi_n$         & intervention profile --- the imposed change\\[3pt]
  $e_t$            & the \textbf{gap}: signed difference, driven to zero\\
\end{tabular}
\end{center}
\vfill
Gradient flows \emph{through the model} into $z_t$: the network Jacobian
spreads the correction over the whole field.
}

\frame{
\frametitle{Guidance schedule}
Applied strength factorizes: $\lambda_t = w \cdot a_t \cdot c_t$
\begin{itemize}
  \item $w$ --- guidance scale: \emph{how hard} to push
  \item $a_t$ --- guidance profile: \emph{when} to push
  \item $c_t$ --- magnitude normalizer: guidance applies the \emph{unit} gradient
        $g_t/\lVert g_t\rVert$; $c_t$ carries the measured size
        ($\lVert g_t\rVert$, Newton magnitude, flow norm, \dots)
\end{itemize}
\vfill
\textbf{Gap-closing adaptive profile:} conceptually close a fraction $\eta$ of the remaining gap each step,
$e_{t+1} = (1-\eta)\, e_t$, such that:
\begin{align*}
  \text{gap}      &:\; e_t = (1-\eta)^{t+1}\, e_0 \quad \text{with } e_0=\eta\\
  \text{profile} &:\; a_t = (1-\eta)^{t+1} = e_t/e_0\\
  \text{closed gap}   &:\; \gamma_t = 1-(1-\eta)^t
\end{align*}
\vfill
\begin{itemize}
  \item $\eta = 1$: whole gap in one step; smaller $\eta$: spread over the flow
  \item Early steps do the heavy lifting; late steps refine.
  \item Different implications depending on guidance method
  \item This is just some heuristic I defined
\end{itemize}
}

\frame{
\frametitle{Guidance methods}
Both steer with $\lambda_t = w \cdot a_t \cdot c_t$ on the same profile $a_t=(1-\eta)^{t+1}$.

\vspace{1em}
They differ only in \emph{how the strength is set}:
\begin{itemize}
  \item \textbf{CLOSE-GAP} --- no $w$: at each step pick the \emph{exact} $\lambda_t$
        that lands the gap on its prescribed path. Self-correcting, one pass.
  \item \textbf{OPTIMIZE-STRENGTH} --- one global scale $w$, \emph{tuned} against the
        final loss by an outer optimization; the profile $a_t$ is fixed.
\end{itemize}
}

\frame{
\frametitle{CLOSE-GAP}
No scale $w$ --- at every step choose $\lambda_t$ to close the gap \emph{exactly}.
\begin{itemize}
  \item Aim at the prescribed remaining gap:
        $e_t^{\text{target}} = (1-\eta)^{t+1}\, e_0$
  \item Land there in one move along $g_t=\nabla_{z_t}\mathcal{L}$:
        \begin{equation*}
          \lambda_t = \frac{2\, e_t \left( e_t - e_t^{\text{target}} \right)}{h_t\, \lVert g_t \rVert^2}
        \end{equation*}
  \item Anchored to the \emph{measured} gap $e_t$ each step \\$\rightarrow$
        drift (model momentum) self-corrects at the next step
\end{itemize}
\vfill
\textbf{Trade-off:} follows the profile $a_t$ exactly, but generates a highly oscillatory behaviour, unless $a_t$ is tuned to minimize the oscillations.
\vfill
Cost: a single guided flow with a gradient pass per step $\sim\!2T$
}


\frame{
\frametitle{OPTIMIZE-STRENGTH}
Fix the profile $a_t$; tune scale $w$ against final loss at step $T$:
\begin{itemize}
  \item Run the whole guided flow at constant $w$: $\lambda_t = w \cdot a_t$
  \item Compute final loss as function of $w$:
        \[
        \mathcal{L}_{\text{final}}(w) = \mathcal{L}\bigl(x^{\text{gui}}_{T}(w)\bigr)
        = \Bigl( M\bigl(x^{\text{gui}}_{T}(w)\bigr) - (1+\phi_n)\,M(x^{\text{ref}}) \Bigr)^{2}
        \]
  \item Iteratively solve $\frac{dL_{\text{final}}}{dw} = 0$ by the secant method 
  \item Stop when the final loss is closed, then rerun once at $w^{*}$
  \item Note: the realized path does \emph{not} follow the prescribed $a_t$
\end{itemize}
\vfill
\textbf{Trade-off:} reaches the target with less oscillatory behaviour at the cost of the outer optimization over $w$
\vfill
Cost: $K$ full flow evaluations with a gradient pass per step $\sim\!K \cdot 2T$.
}

\begin{itemize}
  \item Because we are targeting a specific mask, our analysis should consider the mask in particular.
  \item In fact the guided trajectory has now lost its global spatial coherence (this we specifically aknoledge, and is part of the practical design of the algorithm), but we hope that inside the mask (more specifically inside the box, or near to the center of the mask) things are locally coherent.
  \item This is in contrast to methodologies that try to achieve the same change but with global coherent trajectories.
\end{itemize}

\frame{
\frametitle{Limitations}
\begin{itemize}
  \item \textbf{Fixed mask} --- the region is placed by hand and held still: well suited
        to stationary heatwaves, but not to cyclones or other moving events.
  \item \textbf{Hand-designed target} --- the mask and the change trajectory are ours to
        design, and the loss says nothing about their physical plausibility; realism must
        be validated separately --- for the inputs and the generated states alike.
  \item \textbf{Indirect control} --- loss-based guidance nudges the flow toward a target
        rather than sampling the model's own transition, so the learned dynamics are
        respected only indirectly.
\end{itemize}
\vfill
$\rightarrow$ Sampling-based guidance would address all three at once --- no hand-placed
mask, no prescribed trajectory, no separate realism check --- at the cost of heavier
sampling and its own trade-offs.
}

  \item A more comprehensive \textbf{analysis of physical realism}, with insights into its
        different perspectives
\end{itemize}
